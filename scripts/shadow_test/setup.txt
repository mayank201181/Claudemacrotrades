import glob, json, os, sys
W = sys.argv[1]
def strings(o):
    if isinstance(o, str):
        yield o
    elif isinstance(o, dict):
        for v in o.values():
            yield from strings(v)
    elif isinstance(o, list):
        for v in o:
            yield from strings(v)
marks = ("#### " + "EV.PY START", "#### " + "EV.PY END", "#### " + "BRIEF START", "#### " + "BRIEF END")
def cut(s):
    out = []
    for a, b in ((marks[0], marks[1]), (marks[2], marks[3])):
        i, j = s.find(a + "\n"), s.find("\n" + b)
        if not 0 <= i < j:
            return None
        out.append(s[i + len(a) + 1:j] + "\n")
    return out if out[0].startswith("#!/usr/bin/env python3") else None
best = None
for p in set(glob.glob(os.path.expanduser("~/.claude/projects/*/*.jsonl")) + glob.glob("/root/.claude/projects/*/*.jsonl")):
    for line in open(p, encoding="utf-8", errors="replace"):
        if marks[0] in line and marks[3] in line:
            try:
                for s in strings(json.loads(line)):
                    got = cut(s)
                    if got and (best is None or len(got[0]) + len(got[1]) > len(best[0]) + len(best[1])):
                        best = got
            except ValueError:
                pass
if best:
    for name, text in zip(("ev.py", "brief.md"), best):
        open(os.path.join(W, name), "w", encoding="utf-8").write(text)
print("copied" if best else "not copied")
PYEOF
